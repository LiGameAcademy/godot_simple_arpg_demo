**关键修改**：

2. **修改 _can_cast 方法**：
```gdscript
func _can_cast(ability_comp: Node, target: Node, context: AbilityContext) -> bool:
    # 检查是否有 ActiveFeature
    var has_active_feature = _has_feature(ActiveFeature)
    if not has_active_feature:
        return false  # 没有 ActiveFeature，不能施放
    
    # 调用所有特性的 can_cast
    for feature in features:
        if not is_instance_valid(feature):
            continue
        if not feature.can_cast(self, context):
            return false
    
    return true
```

func _has_feature(feature_class: Variant) -> bool:

### 4. 整合冷却和消耗（可选）

**创建 CooldownFeature 和 CostFeature**：

```gdscript
class_name CooldownFeature extends AbilityFeature
    @export var cooldown_duration: float = 0.0
    var _cooldown_remaining: float = 0.0
    
    func can_cast(ability: GameplayAbility, context: AbilityContext) -> bool:
        return _cooldown_remaining <= 0.0
    
    func commit_cast(ability: GameplayAbility, context: AbilityContext) -> bool:
        if context.skip_cooldown:
            return true
        _cooldown_remaining = cooldown_duration
        return true
    
    func update(ability: GameplayAbility, delta: float) -> void:
        if _cooldown_remaining > 0.0:
            _cooldown_remaining -= delta
            if _cooldown_remaining <= 0.0:
                _cooldown_remaining = 0.0
                # 发出冷却完成事件（通过 AbilityEventBus）
                AbilityEventBus.trigger_game_event(&"ability_cooldown_ready", {
                    "entity": ability.ability_comp.get_parent(),
                    "ability_id": ability.ability_id
                })
```

## 使用示例

### 新方式（完全扁平化）

```gdscript
# 主动技能
var active_ability = GameplayAbility.new()
active_ability.ability_id = &"fireball"
active_ability.statuses = [damage_status]
active_ability.features = [
    ActiveFeature.new(),
    CooldownFeature.new(),
    CostFeature.new(),
    ProjectileFeature.new()
]

# 被动技能
var passive_ability = GameplayAbility.new()
passive_ability.ability_id = &"strength_buff"
passive_ability.statuses = [buff_status]
passive_ability.features = [
    PassiveFeature.new()
]

# 混合技能（学习时获得被动效果，也可以主动施放）
var hybrid_ability = GameplayAbility.new()
hybrid_ability.ability_id = &"blessing"
hybrid_ability.statuses = [buff_status, heal_status]
hybrid_ability.features = [
    ActiveFeature.new(),
    PassiveFeature.new(),
    CooldownFeature.new()
]
```

### 旧方式（向后兼容）

```gdscript
# 仍然可以使用 ActiveAbility 和 PassiveAbility
var active_ability = ActiveAbility.new()
active_ability.ability_id = &"fireball"
# 内部会自动添加 ActiveFeature

var passive_ability = PassiveAbility.new()
passive_ability.ability_id = &"strength_buff"
# 内部会自动添加 PassiveFeature
```

## 注意事项

1. **类型安全**：完全扁平化会降低类型安全，需要在运行时检查特性
2. **配置复杂**：需要配置多个特性才能实现基本功能
3. **学习曲线**：需要理解特性系统才能使用
4. **性能影响**：特性遍历的开销通常可忽略，但需要注意
